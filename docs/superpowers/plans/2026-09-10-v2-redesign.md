# murph.rip v2 Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port the approved static preview in `docs/redesign-previews/r4/` into the real Astro site on branch `v2`, with one content pipeline, plain CSS, and the preview's client modules.

**Architecture:** Astro 5 static site. A `Base.astro` layout renders the sidebar shell, footer strip, background canvas, and settings popover on every page; pages are thin. Content comes from Astro content collections rendered with `render()`; course structure stays folder-based via `_header.md` / `_category.md`. Client behavior is a set of vanilla TypeScript modules under `src/scripts/`, ported function-for-function from the preview's `site.js`, `pages.js`, `critters.js`, `walker.js`, `sidecat.js`, all driven by a `tweakchange` event and CSS variables.

**Tech Stack:** Astro ^5.16, @astrojs/mdx, @astrojs/react (MDX islands only), @fontsource/ibm-plex-sans, @fontsource/ibm-plex-mono, shiki via Astro with the `css-variables` theme, pnpm, GitHub Pages workflow (unchanged).

**Spec:** `docs/superpowers/specs/2026-09-10-v2-redesign-design.md`

**Reference build (read it, port from it):** `docs/redesign-previews/r4/` — `index.html`, `articles.html`, `eecs280.html`, `article.html`, `resume.html`, `resources.html`, `snippets.html`, `style.css`, `pages.css`, `resume.css`, `walker.css`, `site.js`, `pages.js`, `critters.js`, `walker.js`, `sidecat.js`, `logo-a.svg`, `logo-b.svg`, `logo-c.svg`, `spec.md`, `spec-v2.md`, `spec-resume.md`. Content facts: `docs/redesign-previews/_brief/content.md` and `_brief/resume.md`.

## Global Constraints

- Branch `v2`. Never merge to `main` in this plan. Commit after every task.
- Node 22, pnpm 10 (the deploy workflow's versions). `pnpm build` must pass with zero content-collection warnings after every task.
- Article URLs stay identical to today: `/articles/<filename-without-extension>` with the filename's original casing (for example `/articles/complexity_analysis_big_Oshit`) and `/articles/<category-slug>` for course pages (`/articles/eecs280`).
- No light mode. All palettes are dark: `ember` (default), `alley`, `space`. No `prefers-color-scheme` handling anywhere.
- Every animation loop: pauses on `document.hidden`, draws one static frame under `prefers-reduced-motion: reduce`, reads colors from CSS variables, re-reads on `tweakchange`, allocates nothing per frame.
- Fonts: IBM Plex Sans (body, headings) and IBM Plex Mono (meta, labels, code) from fontsource only. No Google Fonts.
- Content is verbatim from the repo's markdown and from `_brief/resume.md`. Never invent copy.
- Existing Tailwind classes keep working until Task 9 removes Tailwind; do not add new Tailwind classes anywhere.
- Commit messages: Conventional Commits, ending with the two attribution lines used on this branch (see `git log -3`).

```ts
import { tweakDefaults, type Tweaks } from '../site.config';

const KEY = 'tweaks';
export let state: Tweaks = { ...tweakDefaults };

export function readState(): Tweaks {
  try { const s = JSON.parse(localStorage.getItem(KEY) || '{}'); return { ...tweakDefaults, ...s }; }
  catch { return { ...tweakDefaults }; }
}

export function writeAttrs(): void {
  const h = document.documentElement;
  const d = h.dataset as Record<string, string>;
  d.palette = state.palette; d.logo = state.logo; d.mich = state.mich; d.draw = state.draw;
  for (const k of ['orb','bg','strands','motes','paws','walker','deer','raccoon'] as const) d[k] = String(state[k]);
  h.style.setProperty('--bg-opacity', String(state.bgOpacity / 100));
  h.style.setProperty('--bg-dim', String(state.bgDim / 100));
  const icon = document.getElementById('favicon') as HTMLLinkElement | null;
  if (icon && state.logo !== 'orb') icon.href = `/${state.logo.replace('mark-', 'logo-')}.svg`;
}

export function set<K extends keyof Tweaks>(key: K, value: Tweaks[K], persist = true): void {
  state[key] = value;
  if (persist) { try { localStorage.setItem(KEY, JSON.stringify(state)); } catch {} }
  writeAttrs();
  document.dispatchEvent(new CustomEvent('tweakchange', { detail: { key, value, state } }));
}

export function reset(): void {
  state = { ...tweakDefaults };
  try { localStorage.removeItem(KEY); } catch {}
  writeAttrs(); sync();
  document.dispatchEvent(new CustomEvent('tweakchange', { detail: { key: 'reset', value: null, state } }));
}

function sync(): void {
  const form = document.getElementById('tweaks');
  if (!form) return;
  form.querySelectorAll<HTMLInputElement | HTMLSelectElement>('[name]').forEach((el) => {
    const k = el.name as keyof Tweaks;
    if (!(k in state)) return;
    if (el instanceof HTMLInputElement && el.type === 'checkbox') el.checked = !!state[k];
    else el.value = String(state[k]);
  });
}

export function initTweaks(): void {
  state = readState();
  const qs = new URLSearchParams(location.search);
  for (const k of ['palette', 'logo', 'draw'] as const) { const v = qs.get(k); if (v) (state as any)[k] = v; }
  (window as any).TWEAKS = state;
  writeAttrs(); sync();
  const form = document.getElementById('tweaks');
  form?.addEventListener('input', (e) => {
    const el = e.target as HTMLInputElement | HTMLSelectElement;
    const k = el.name as keyof Tweaks;
    if (!(k in state)) return;
    const v = el instanceof HTMLInputElement && el.type === 'checkbox' ? (el.checked ? 1 : 0)
      : el instanceof HTMLInputElement && el.type === 'range' ? Number(el.value) : el.value;
    set(k, v as Tweaks[typeof k]);
  });
  form?.querySelector('.tweak__reset')?.addEventListener('click', (e) => { e.preventDefault(); reset(); });
}
```

```ts
import { getCollection, type CollectionEntry } from 'astro:content';
import { getBlogStructure, type Category, type Header } from './content';

export interface Post {
  entry: CollectionEntry<'blog'>; slug: string; headerSlug: string; categorySlug: string;
  category: Category; header: Header; date: Date; words: number; minutes: number;
}

function countWords(body: string | undefined): number {
  return (body || '').replace(/```[\s\S]*?```/g, ' ').split(/\s+/).filter(Boolean).length;
}

let cache: Post[] | null = null;
export async function getPosts(): Promise<Post[]> {
  if (cache) return cache;
  const structure = getBlogStructure();
  const entries = await getCollection('blog');
  const posts: Post[] = [];
  for (const entry of entries) {
    const [headerSlug, categorySlug, ...rest] = entry.id.split('/');
    const header = structure.find((h) => h.slug === headerSlug);
    const category = header?.categories.find((c) => c.slug === categorySlug);
    if (!header || !category || rest.length !== 1) continue; // files outside header/category folders are ignored
    const words = countWords(entry.body);
    posts.push({ entry, slug: rest[0], headerSlug, categorySlug, category, header, date: entry.data.date, words, minutes: Math.max(1, Math.round(words / 220)) });
  }
  posts.sort((a, b) => b.date.getTime() - a.date.getTime() || (b.entry.data.order || 0) - (a.entry.data.order || 0));
  cache = posts;
  return posts;
}

export async function getPost(slug: string): Promise<Post | undefined> {
  return (await getPosts()).find((p) => p.slug === slug);
}

export async function getCategoryPosts(categorySlug: string): Promise<Post[]> {
  return (await getPosts()).filter((p) => p.categorySlug === categorySlug)
    .sort((a, b) => a.date.getTime() - b.date.getTime() || (a.entry.data.order || 999) - (b.entry.data.order || 999));
}

export async function getStats() {
  const posts = await getPosts();
  return { total: posts.length, words: posts.reduce((n, p) => n + p.words, 0), since: Math.min(...posts.map((p) => p.date.getFullYear())) };
}

export async function courseCounts(): Promise<Record<string, number>> {
  const counts: Record<string, number> = {};
  for (const p of await getPosts()) counts[p.categorySlug] = (counts[p.categorySlug] || 0) + 1;
  return counts;
}

export const fmtDate = (d: Date) => d.toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric', timeZone: 'UTC' });
export const isoDate = (d: Date) => d.toISOString().slice(0, 10);
```
